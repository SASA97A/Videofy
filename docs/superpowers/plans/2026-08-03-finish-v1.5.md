`Video Size Optimizer/Services/Codecs/CodecAudioMapper.cs`:

```csharp
using System.Collections.Generic;
using System.Linq;
using Video_Size_Optimizer.Models;

namespace Video_Size_Optimizer.Services.Codecs;

public static class CodecAudioMapper
{
    private static readonly Dictionary<string, AudioCodecOption> OptionByCodec =
        AudioCodecOption.CreateDefaults()
            .Where(o => !o.IsPassthrough)
            .ToDictionary(o => o.CodecName, o => o);

    public static List<AudioCodecOption> MapDetectedAudioCodecs(IEnumerable<Codec> detected)
    {
        var result = detected
            .Where(c => c.Type == CodecType.Audio && OptionByCodec.ContainsKey(c.Name))
            .Select(c => OptionByCodec[c.Name])
            .Distinct()
            .ToList();

        if (result.All(o => !o.IsPassthrough))
            result.Add(AudioCodecOption.CreateDefaults().Last());

        return result;
    }
}
```

    private async Task LoadAudioCodecsAsync()
    {
        try
        {
            var codecService = new CodecService();
            var detected = await codecService.GetAudioCodecsAsync();
            if (detected.Count == 0)
                return;

            var mapped = CodecAudioMapper.MapDetectedAudioCodecs(detected);
            if (mapped.Count == 0)
                return;

            AvailableAudioCodecs.Clear();
            foreach (var option in mapped)
                AvailableAudioCodecs.Add(option);

            if (SelectedAudioCodec == null || !AvailableAudioCodecs.Contains(SelectedAudioCodec))
                SelectedAudioCodec = AvailableAudioCodecs.FirstOrDefault(o => o.CodecName == "aac")
                                     ?? AvailableAudioCodecs.FirstOrDefault();
        }
        catch (Exception ex)
        {
            LogService.Instance.Log($"Audio codec load failed: {ex.Message}", LogLevel.Warning, "Codecs");
        }
    }

using Video_Size_Optimizer.Models;
using Video_Size_Optimizer.Utils;

namespace Video_Size_Optimizer.Services.Profiles;

public sealed class ProfileFillResult
{
    public int CrfValue { get; init; } = 28;
    public string Resolution { get; init; } = "Original Resolution";
    public string Fps { get; init; } = "Original";
    public string? Encoder { get; init; }
    public string AudioCodecName { get; init; } = "aac";
    public bool StripMetadata { get; init; } = true;
    public string OutputFormat { get; init; } = ".mp4";
}

public static class ProfileSettingsMapper
{
    public static ProfileFillResult Map(EncodingProfile profile) => new()
    {
        CrfValue = profile.CrfValue,
        Resolution = MapResolution(profile.Resolution),
        Fps = profile.Framerate <= 0 ? "Original" : profile.Framerate.ToString(),
        Encoder = FindEncoderDisplayName(profile.VideoCodec),
        AudioCodecName = profile.AudioCodec,
        StripMetadata = profile.StripMetadata,
        OutputFormat = string.IsNullOrEmpty(profile.OutputFormat) ? ".mp4" : profile.OutputFormat
    };

    public static string MapResolution(string profileResolution) => profileResolution switch
    {
        "2160p" => "4K (3840p)",
        "1440p" => "2K (1440p)",
        "1080p" => "Full HD (1080p)",
        "720p" => "HD (720p)",
        "480p" => "SD (480p)",
        "360p" => "Mobile (360p)",
        _ => "Original Resolution"
    };

    public static string? FindEncoderDisplayName(string codecName)
    {
        foreach (var kv in AppConstants.EncoderMap)
        {
            if (kv.Value == codecName)
                return kv.Key;
        }
        return null;
    }
}
```

using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;

namespace Video_Size_Optimizer.Utils;

public enum DroppedItemKind
{
    None,
    Folder,
    SupportedFile,
    UnsupportedFile
}

public static class DroppedItemClassifier
{
    public static DroppedItemKind Classify(string path, IEnumerable<string> supportedExtensions)
    {
        if (string.IsNullOrWhiteSpace(path))
            return DroppedItemKind.None;

        if (Directory.Exists(path))
            return DroppedItemKind.Folder;

        if (!File.Exists(path))
            return DroppedItemKind.None;

        string ext = Path.GetExtension(path);
        return supportedExtensions.Any(e => string.Equals(e, ext, StringComparison.OrdinalIgnoreCase))
            ? DroppedItemKind.SupportedFile
            : DroppedItemKind.UnsupportedFile;
    }
}
```
